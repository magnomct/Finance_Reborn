                  onChange={e => setSearchQuery(e.target.value)}
                  className={`w-full pl-9 pr-4 py-1.5 rounded-xl text-xs ${theme.inputBg} focus:outline-none focus:ring-2 focus:ring-blue-500`}
                />
              </div>
            </div>

            {/* Tabela de Extratos */}
            <div className={`${theme.card} rounded-2xl overflow-hidden`}>
              <div className="overflow-x-auto">
                <table className="w-full text-left text-xs border-collapse">
                  <thead>
                    <tr className="border-b border-slate-100 dark:border-slate-800 text-slate-400 uppercase text-[10px] font-bold">
                      <th className="py-3 px-3">Data</th>
                      <th className="py-3 px-3">Descrição</th>
                      <th className="py-3 px-3">Centro</th>
                      <th className="py-3 px-3">Conta</th>
                      <th className="py-3 px-3">Pote</th>
                      <th className="py-3 px-3 text-right">Valor (R$)</th>
                      <th className="py-3 px-3 text-center">Ações</th>
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-slate-100 dark:divide-slate-800/80">
                    {filteredTransactions.map((tx) => {
                      const isPositive = tx.type === 'receita' || tx.type === 'reembolso';
                      return (
                        <tr key={tx.id} className={`${theme.hoverRow} transition`}>
                          <td className="py-3 px-3 text-slate-400 font-mono">{tx.date}</td>
                          <td className="py-3 px-3 font-semibold text-slate-900 dark:text-white">{tx.description}</td>
                          <td className="py-3 px-3 text-slate-500">{tx.costCenter}</td>
                          <td className="py-3 px-3 text-slate-500">{tx.account}</td>
                          <td className="py-3 px-3">
                            <span className="px-2 py-0.5 rounded-md bg-slate-100 dark:bg-slate-800 text-slate-600 dark:text-slate-300 font-medium">
                              {tx.jar}
                            </span>
                          </td>
                          <td className={`py-3 px-3 text-right font-black ${isPositive ? 'text-emerald-600 dark:text-emerald-400' : 'text-slate-800 dark:text-rose-400'}`}>
                            {isPositive ? '+' : '-'} R$ {Math.abs(tx.amount).toFixed(2)}
                          </td>
                          <td className="py-3 px-3 text-center">
                            <button 
                              onClick={() => handleDeleteTransaction(tx.id)}
                              className="text-slate-400 hover:text-rose-500 transition p-1"
                              title="Excluir"
                            >
                              <Trash2 className="w-3.5 h-3.5" />
                            </button>
                          </td>
                        </tr>
                      );
                    })}
                  </tbody>
                </table>
              </div>
            </div>
          </div>
        )}

        {/* ============================================================== */}
        {/* TELA 6: OPERAÇÃO PJ (DRE Simplificado & Distribuição de Lucros) */}
        {/* ============================================================== */}
        {activeScreen === 'pj' && (
          <div className="px-4 space-y-4">
            <div className="flex justify-between items-center">
              <div>
                <h2 className="text-lg font-bold text-amber-600 dark:text-amber-400 flex items-center gap-2">
                  <Building2 className="w-5 h-5" />
                  Demonstrativo da Operação PJ (Inter PJ)
                </h2>
                <p className="text-xs text-slate-400">Faturamento de R$ 3.060,86 quitando convênio médico e DAS</p>
              </div>
              <span className="px-3 py-1 bg-amber-100 text-amber-800 dark:bg-amber-950 dark:text-amber-300 font-bold text-xs rounded-xl">
                Margem Líquida: {financialMetrics.pjProfitMargin.toFixed(1)}%
              </span>
            </div>

            <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
              <div className={`${theme.card} p-4 rounded-2xl`}>
                <span className="text-[10px] text-slate-400 uppercase font-bold">1. Faturamento Bruto PJ</span>
                <p className="text-xl font-black text-emerald-600 dark:text-emerald-400 mt-1">
                  +R$ {financialMetrics.pjServiceIncome.toFixed(2)}
                </p>
                <span className="text-[11px] text-slate-400">Entrada recebida na conta Inter PJ</span>
              </div>

              <div className={`${theme.card} p-4 rounded-2xl`}>
                <span className="text-[10px] text-slate-400 uppercase font-bold">2. Custos Corporativos PJ</span>
                <p className="text-xl font-black text-rose-500 dark:text-rose-400 mt-1">
                  -R$ {financialMetrics.pjExpenses.toFixed(2)}
                </p>
                <span className="text-[11px] text-slate-400">Convênio (R$ 1.296,45) + DAS (R$ 172,10)</span>
              </div>

              <div className={`${theme.card} p-4 rounded-2xl border-l-4 border-l-amber-500`}>
                <span className="text-[10px] text-amber-600 dark:text-amber-400 uppercase font-bold">3. Lucro Líquido Livre PJ</span>
                <p className="text-xl font-black text-amber-600 dark:text-amber-400 mt-1">
                  +R$ {financialMetrics.pjNetProfit.toFixed(2)}
                </p>
                <span className="text-[11px] text-slate-400">Saldo 100% disponível para PF</span>
              </div>
            </div>
          </div>
        )}

        {/* ============================================================== */}
        {/* TELA 7: RELATÓRIOS & DIAGNÓSTICO (Visão Estratégica)             */}
        {/* ============================================================== */}
        {activeScreen === 'relatorios' && (
          <div className="px-4 space-y-4">
            <h2 className="text-lg font-bold text-slate-900 dark:text-white">Diagnóstico Geral das Finanças</h2>
            <div className={`${theme.card} p-5 rounded-2xl space-y-3`}>
              <div className="flex items-center justify-between">
                <span className="text-xs font-bold uppercase text-slate-400">Status Geral do Orçamento</span>
                <span className="px-2.5 py-1 rounded-full bg-emerald-100 text-emerald-800 dark:bg-emerald-950 dark:text-emerald-300 font-bold text-xs">
                  Equilíbrio Orçamentário Atingido
                </span>
              </div>

              <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 pt-2">
                <div className="p-3 bg-slate-50 dark:bg-slate-800 rounded-xl">
                  <span className="text-[10px] text-slate-400 block uppercase">Salário CLT</span>
                  <span className="text-sm font-bold text-slate-800 dark:text-white">R$ 2.060,00</span>
                </div>
                <div className="p-3 bg-slate-50 dark:bg-slate-800 rounded-xl">
                  <span className="text-[10px] text-slate-400 block uppercase">Lucro Líquido PJ</span>
                  <span className="text-sm font-bold text-amber-600 dark:text-amber-400">R$ 1.592,31</span>
                </div>
                <div className="p-3 bg-slate-50 dark:bg-slate-800 rounded-xl">
                  <span className="text-[10px] text-slate-400 block uppercase">Despesas Líquidas</span>
                  <span className="text-sm font-bold text-slate-800 dark:text-white">R$ 3.650,00</span>
                </div>
                <div className="p-3 bg-emerald-50 dark:bg-emerald-950/60 rounded-xl">
                  <span className="text-[10px] text-emerald-600 dark:text-emerald-400 block uppercase font-bold">Saldo Livre</span>
                  <span className="text-sm font-black text-emerald-600 dark:text-emerald-400">+R$ 2,31</span>
                </div>
              </div>
            </div>
          </div>
        )}

      </main>

      {/* ============================================================== */}
      {/* MODAL DE NOVO LANÇAMENTO (Fiel ao Formulário Clássico do Yupee) */}
      {/* ============================================================== */}
      {isModalOpen && (
        <div className="fixed inset-0 bg-black/60 backdrop-blur-sm z-50 flex items-center justify-center p-4">
          <div className={`${theme.card} w-full max-w-md rounded-3xl p-6 shadow-2xl space-y-4 animate-in fade-in zoom-in-95 duration-150`}>
            
            {/* Cabeçalho do Modal */}
            <div className="flex items-center justify-between pb-2 border-b border-slate-100 dark:border-slate-800">
              <div className="flex items-center gap-2">
                <div className="flex items-center gap-0.5">
                  <span className="w-4 h-4 rounded bg-rose-500 text-white font-bold text-[9px] flex items-center justify-center">y</span>
                  <span className="w-4 h-4 rounded bg-amber-500 text-white font-bold text-[9px] flex items-center justify-center">u</span>
                  <span className="w-4 h-4 rounded bg-emerald-500 text-white font-bold text-[9px] flex items-center justify-center">p</span>
                </div>
                <h3 className="font-bold text-sm text-slate-900 dark:text-white">Novo Lançamento</h3>
              </div>
              <button 
                onClick={() => setIsModalOpen(false)}
                className="p-1 rounded-full text-slate-400 hover:bg-slate-100 dark:hover:bg-slate-800"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <form onSubmit={handleAddTransaction} className="space-y-3.5 text-xs">
              
              {/* Segmented Control do Yupee: Tipo de Lançamento */}
              <div>
                <label className="text-[11px] font-bold text-slate-500 block mb-1">Tipo de Lançamento</label>
                <div className="grid grid-cols-3 gap-1.5 p-1 bg-slate-100 dark:bg-slate-800 rounded-xl">
                  <button
                    type="button"
                    onClick={() => setType('despesa')}
                    className={`py-1.5 rounded-lg font-bold transition ${
                      type === 'despesa' ? 'bg-rose-500 text-white shadow-sm' : 'text-slate-500 hover:text-slate-800 dark:hover:text-white'
                    }`}
                  >
                    Pagamento (-)
                  </button>
                  <button
                    type="button"
                    onClick={() => setType('receita')}
                    className={`py-1.5 rounded-lg font-bold transition ${
                      type === 'receita' ? 'bg-emerald-500 text-white shadow-sm' : 'text-slate-500 hover:text-slate-800 dark:hover:text-white'
                    }`}
                  >
                    Recebimento (+)
                  </button>
                  <button
                    type="button"
                    onClick={() => setType('reembolso')}
                    className={`py-1.5 rounded-lg font-bold transition ${
                      type === 'reembolso' ? 'bg-blue-500 text-white shadow-sm' : 'text-slate-500 hover:text-slate-800 dark:hover:text-white'
                    }`}
                  >
                    Reembolso
                  </button>
                </div>
              </div>

              {/* Data e Valor (Lado a Lado) */}
              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="text-[11px] font-bold text-slate-500 block mb-1">Data</label>
                  <input 
                    type="date"
                    value={txDate}
                    onChange={e => setTxDate(e.target.value)}
                    className={`w-full px-3 py-2 rounded-xl text-xs font-medium ${theme.inputBg} focus:outline-none focus:ring-2 focus:ring-blue-500`}
                  />
                </div>
                <div>
                  <label className="text-[11px] font-bold text-slate-500 block mb-1">Valor (R$)</label>
                  <input 
                    type="number"
                    step="0.01"
                    placeholder="0,00"
                    value={amount}
                    onChange={e => setAmount(e.target.value)}
                    className={`w-full px-3 py-2 rounded-xl text-xs font-bold ${theme.inputBg} focus:outline-none focus:ring-2 focus:ring-blue-500`}
                  />
                </div>
              </div>

              {/* Descrição */}
              <div>
                <label className="text-[11px] font-bold text-slate-500 block mb-1">Descrição</label>
                <input 
                  type="text"
                  placeholder="Ex: Mercado, Uber, Salário, Internet"
                  value={desc}
                  onChange={e => setDesc(e.target.value)}
                  className={`w-full px-3 py-2 rounded-xl text-xs ${theme.inputBg} focus:outline-none focus:ring-2 focus:ring-blue-500`}
                />
              </div>

              {/* Pote / Categoria dos 6 Potes */}
              <div>
                <label className="text-[11px] font-bold text-slate-500 block mb-1">Pote / Grupo Orçamentário</label>
                <select
                  value={jar}
                  onChange={e => setJar(e.target.value as JarType)}
                  className={`w-full px-3 py-2 rounded-xl text-xs ${theme.inputBg} focus:outline-none focus:ring-2 focus:ring-blue-500`}
                >
                  <option value="NEC_55">55% Necessidades Básicas (Contas, Casa, Transporte)</option>
                  <option value="DIV_10">10% Diversão & Estilo (Lazer, Passeios, C6)</option>
                  <option value="PLP_10">10% Reserva & Sonhos (Poupança a Longo Prazo)</option>
                  <option value="LIF_10">10% Liberdade Financeira (Investimentos)</option>
                  <option value="EDU_10">10% Educação Continuada (Cursos e Livros)</option>
                  <option value="FER_5">5% Férias & Doações</option>
                  <option value="PJ_EXCLUIDO">PJ - Operação da Empresa</option>
                </select>
              </div>

              {/* Conta e Centro de Custos */}
              <div className="grid grid-cols-2 gap-3">
                <div>
                  <label className="text-[11px] font-bold text-slate-500 block mb-1">Conta de Pagamento</label>
                  <select
                    value={account}
                    onChange={e => setAccount(e.target.value as AccountType)}
                    className={`w-full px-3 py-2 rounded-xl text-xs ${theme.inputBg} focus:outline-none focus:ring-2 focus:ring-blue-500`}
                  >
                    <option value="Inter PF">Inter PF</option>
                    <option value="Inter PJ">Inter PJ</option>
                    <option value="C6">Cartão C6</option>
                    <option value="Passe">Passe / Transporte</option>
                    <option value="Outro">Outro</option>
                  </select>
                </div>

                <div>
                  <label className="text-[11px] font-bold text-slate-500 block mb-1">Centro de Custos</label>
                  <select
                    value={costCenter}
                    onChange={e => setCostCenter(e.target.value as CostCenter)}
                    className={`w-full px-3 py-2 rounded-xl text-xs ${theme.inputBg} focus:outline-none focus:ring-2 focus:ring-blue-500`}
                  >
                    <option value="Magno">Magno</option>
                    <option value="Nathan">Nathan</option>
                    <option value="PJ">PJ</option>
                  </select>
                </div>
              </div>

              {/* Botões do Rodapé do Formulário (Salvar / Cancelar Yupee) */}
              <div className="pt-3 flex items-center justify-end gap-2 border-t border-slate-100 dark:border-slate-800">
                <button
                  type="button"
                  onClick={() => setIsModalOpen(false)}
                  className="px-4 py-2 rounded-xl font-semibold text-slate-500 hover:bg-slate-100 dark:hover:bg-slate-800 transition"
                >
                  Cancelar
                </button>
                <button
                  type="submit"
                  className="px-5 py-2 rounded-xl bg-blue-500 hover:bg-blue-600 text-white font-bold shadow-md transition active:scale-95"
                >
                  Salvar Lançamento
                </button>
              </div>

            </form>
          </div>
        </div>
      )}

    </div>
  );
}
